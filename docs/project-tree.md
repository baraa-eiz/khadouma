# ملحق 1: شجرة مجلدات وملفات النظام (Project Directory Tree)

يوفر هذا المخطط هيكل وتفرعات ملفات منصة **خدومة** بشكل مرئي لتسهيل تتبع الملفات.

```text
/home/cnc-jordan-service/htdocs/service.cnc-jordan.com/
├── README.md                          # ملف التعريف العام والتشغيل
├── index.php                          # المتحكم الرئيسي للموقع (Front Controller)
├── admin/                             # لوحة التحكم الإدارية للمشرفين
│   ├── dashboard.php                  # شاشة الإحصائيات العامة
│   ├── login.php                      # شاشة تسجيل الدخول المؤمنة
│   └── logout.php                     # الخروج وتصفية جلسة المشرف
├── app/                               # نواة المنصة والمنطق البرمجي
│   ├── Core/                          # فئات النواة البرمجية الأساسية
│   │   ├── Database.php               # مدير اتصال PDO المجهز
│   │   ├── Response.php               # ترويسات ومخرجات JSON والتحويل
│   │   ├── Router.php                 # محرك توجيه وحل المسارات
│   │   └── Validator.php              # فحص حقول البيانات ومطابقة المدخلات
│   ├── Helpers/                       # الدوال المساعدة المشتركة
│   │   ├── image_helpers.php          # توليد ومعالجة صور SVG البديلة
│   │   ├── security_helpers.php       # معالجة XSS والوقاية من CSRF
│   │   ├── seo_helpers.php            # عناوين وأوصاف محركات البحث
│   │   ├── text_helpers.php           # تنظيف وتطبيع الحروف العربية
│   │   └── url_helpers.php            # تكوين روابط المسارات المطلقة
│   └── Repositories/                  # مستودعات عزل استعلامات قاعدة البيانات
│       ├── AdminUserRepository.php    # مستودع حسابات الإدارة
│       ├── AreaRepository.php         # مستودع الأحياء السكنية
│       ├── CityRepository.php         # مستودع المحافظات السورية
│       ├── ProviderRepository.php     # مستودع تفاصيل وألبومات الحرفيين
│       ├── ReportRepository.php       # مستودع بلاغات وشكاوى العملاء
│       ├── ReviewRepository.php       # مستودع تقييمات العملاء
│       └── SettingsRepository.php     # مستودع إعدادات وتكوين النظام
├── config/                            # ملفات الإعدادات والربط
│   ├── config.php                     # ثوابت وإعدادات استكشاف البيئة تلقائياً
│   └── database.php                   # إعدادات معايير اتصال PDO
├── database/                          # مجلد الهيكلية وتتبع التغييرات
│   ├── backups/                       # مجلد حفظ نسخ قواعد البيانات الاحتياطية
│   ├── migrations/                    # سجل تتبع التعديلات والترقيات
│   │   └── 0001_initial_schema.sql    # النسخة التأسيسية الأولى للجداول الـ 19
│   ├── notes.md                       # ملاحظات هيكلية وترميز الجداول
│   ├── reset_database.sql             # ملف تهيئة الجداول وتنقيتها بالكامل
│   ├── schema.sql                     # الهيكل الأساسي للجداول
│   └── seed.sql                       # البيانات التجريبية الأولية ومستخدم الإدارة
├── docs/                              # توثيقات فنية متكاملة للمشروع
│   ├── admin-control-plan.md          # خطة عمل وتطوير واجهات CRUD القادمة
│   ├── admin-recovery.md              # ملحق 2: كود استعادة كلمة مرور المشرف
│   ├── architecture-decisions.md      # ملحق 1: مبررات اختيار التقنيات والأنماط
│   ├── architecture.md                # توثيق البنية والدورة البرمجية
│   ├── backup-strategy.md             # ملحق 2: إستراتيجية نسخ قواعد البيانات الآلية
│   ├── code-index.md                  # ملحق 1: فهرس الفئات والدوال المساعدة
│   ├── code-map.md                    # خارطة الملفات التفصيلية ووظائفها
│   ├── database-schema.md             # قاموس وتعريفات فهارس جداول قاعدة البيانات
│   ├── future-stages.md               # خارطة الطريق للمراحل التطويرية اللاحقة
│   ├── image-policy.md                # سياسة معالجة وحظر صور الذكاء الاصطناعي
│   ├── image-sources.md               # ملحق 1: دليل روابط ومصادر الصور البديلة
│   ├── project-tree.md                # ملحق 1: شجرة مجلدات وملفات النظام (هذا الملف)
│   └── setup-xampp.md                 # دليل تشغيل المنصة محلياً على XAMPP
├── includes/                          # المكونات المشتركة لواجهات الـ HTML
│   ├── auth.php                       # بوابة التحقق ومنع غير المشرفين
│   ├── footer.php                     # تذييل الصفحات المشترك
│   ├── header.php                     # ترويسة وقوائم التصفح العامة
│   ├── init.php                       # مهيئ الجلسات ومحفز الإقلاع
│   └── layout.php                     # المنسق العام للقوالب المشتركة
├── pages/                             # واجهات العرض العامة
│   ├── about-us.php                   # صفحة من نحن الديناميكية
│   ├── home.php                       # الصفحة الرئيسية للمنصة ودليل الحرفيين
│   ├── not-found.php                  # صفحة الخطأ 404 المخصصة
│   ├── privacy.php                    # صفحة سياسة الخصوصية وسرية المعلومات
│   └── terms.php                      # صفحة الشروط والأحكام لاستخدام المنصة
└── public/                            # الملفات الثابتة المعروضة للجمهور
    └── assets/                        # أصول واجهات العرض والوسائط
        ├── css/
        │   └── style.css              # ملف التنسيق المخصص للمنصة (Light Warm Trust)
        ├── js/
        │   └── main.js                # ملف الجافاسكربت للتحسينات التفاعلية
        └── images/
            ├── placeholders/          # مجلد تخزين الصور الافتراضية
            ├── providers/             # مجلد تخزين صور الحسابات للحرفيين
            ├── services/              # مجلد تخزين صور وتصاميم الخدمات
            └── uploads/               # مجلد تخزين الصور المرفوعة من الإدارة
```
